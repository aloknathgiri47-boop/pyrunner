   289→function loadState(): PersistedState | null {
   290→  if (typeof window === 'undefined') return null
   291→  try {
   292→    const raw = window.localStorage.getItem(STORAGE_KEY)
   293→    if (!raw) return null
   294→    const parsed = JSON.parse(raw) as PersistedState
   295→    if (typeof parsed.code !== 'string') return null
   296→    return {
   297→      code: parsed.code,
   298→      language: ['java','c','cpp','r','javascript','php','csharp','dart','flutter'].includes(parsed.language) ? parsed.language : 'python',
   299→    }
   300→  } catch {
   301→    return null
   302→  }
   303→}

   305→function encodeToHash(code: string, language: Language): string {
   306→  const encode = (s: string) => {
   307→    const bytes = new TextEncoder().encode(s)
   308→    let bin = ''
   309→    for (let i = 0; i < bytes.length; i++) bin += String.fromCharCode(bytes[i])
   310→    return btoa(bin).replace(/\+/g, '-').replace(/\//g, '_').replace(/=+$/, '')
   311→  }
   312→  const params = new URLSearchParams()
   313→  params.set('c', encode(code))
   314→  params.set('l', language)
   315→  return `#$\{params.toString()}`
   316→}

   322→function getInitialState(): { code: string; language: Language } {
   323→  if (typeof window === 'undefined') return { code: DEFAULT_CODE, language: 'python' }
   324→  // URL hash takes priority, then localStorage, then default.
   325→  const hash = window.location.hash
   326→  if (hash && hash.length > 2) {
   327→    try {
   328→      const params = new URLSearchParams(hash.slice(1))
   329→      const codeB64 = params.get('c')
   330→      const lang = params.get('l')
   331→      if (codeB64) {
   332→        const decode = (s: string) => {
   333→          const padded = s.replace(/-/g, '+').replace(/_/g, '/')
   334→          const pad = padded.length % 4 === 0 ? '' : '='.repeat(4 - (padded.length % 4))
   335→          const bin = atob(padded + pad)
   336→          const bytes = new Uint8Array(bin.length)
   337→          for (let i = 0; i < bin.length; i++) bytes[i] = bin.charCodeAt(i)
   338→          return new TextDecoder('utf-8').decode(bytes)
   339→        }
   340→        return {
   341→          code: decode(codeB64),
   342→          language: ['java','c','cpp','r','javascript','php','csharp','dart','flutter'].includes(lang) ? lang : 'python',
   343→        }
   344→      }
   345→    } catch {
   346→      /* fall through */
   347→    }
   348→  }
   349→  const persisted = loadState()
   350→  if (persisted) {
   351→    return { code: persisted.code, language: persisted.language }
   352→  }
   353→  return { code: DEFAULT_CODE, language: 'python' }
   354→}

   356→export default function Home() {
   357→  const { setTheme, resolvedTheme } = useTheme()
   358→  // Lazy initializer: runs once on the client during the very first render.
   359→  // On the server it returns DEFAULT_CODE (window is undefined). This avoids
   360→  // any useEffect-based hydration that would trip react-hooks/set-state-in-effect.
   361→  // suppressHydrationWarning on <html> covers the resulting markup difference.

   364→  const [language, setLanguage] = useState<Language>(initialState.language)
   365→  const [activeExampleId, setActiveExampleId] = useState<string | null>(null)
   366→
   367→  const [chunks, setChunks] = useState<OutputChunk[]>([])
   368→  const [isRunning, setIsRunning] = useState(false)
   369→  const [result, setResult] = useState<RunResult | null>(null)
   370→  const [copied, setCopied] = useState(false)
   371→  const [shared, setShared] = useState(false)
   372→  const [inputValue, setInputValue] = useState('')
   373→  const [awaitingInput, setAwaitingInput] = useState(false)
   374→  const [stdinText, setStdinText] = useState('')
   375→  const [showStdin, setShowStdin] = useState(false)
   376→  const [flutterPort, setFlutterPort] = useState<number | null>(null)
   377→
   378→  const socketRef = useRef<Socket | null>(null)
   379→  const chunkIdRef = useRef(0)
   380→  const consoleEndRef = useRef<HTMLDivElement | null>(null)
   381→  const inputRef = useRef<HTMLInputElement | null>(null)
   382→  const isRunningRef = useRef(false)
   383→  const languageRef = useRef<Language>(language)
   384→
   385→  // No hydration effect needed — the lazy initializer handles it.
   386→
   387→  // Keep languageRef in sync so the socket callbacks can read the current value
   388→  useEffect(() => {
   389→    languageRef.current = language
   390→  }, [language])
   391→
   392→  // ---- Persist code + language (debounced) ----
   393→  useEffect(() => {
   394→    const t = setTimeout(() => {
   395→      try {
   396→        window.localStorage.setItem(
   397→          STORAGE_KEY,
   398→          JSON.stringify({ code, language } satisfies PersistedState),
   399→        )
   400→      } catch {
   401→        /* ignore */
   402→      }
   403→    }, 400)
   404→    return () => clearTimeout(t)
   405→  }, [code, language])
   406→
   407→  // ---- WebSocket connection (lazy: only connect when running) ----
   408→  const ensureSocket = useCallback((): Socket => {
   409→    if (socketRef.current) return socketRef.current
   410→    // Use polling+websocket transports so the XTransformPort query parameter
   411→    // is preserved during the initial HTTP handshake (websocket-only skips
   412→    // the polling handshake and the query gets dropped by the gateway).
   413→    // Enable reconnection so if the runner restarts (watchdog auto-restart),
   414→    // the client automatically reconnects without showing "xhr poll error".
   415→    const sock = io('/?XTransformPort=3003', {
   416→      transports: ['polling', 'websocket'],
   417→      forceNew: true,
   418→      reconnection: true,
   419→      reconnectionAttempts: 10,
   420→      reconnectionDelay: 500,
   421→      reconnectionDelayMax: 3000,
   422→      timeout: 10_000,
   423→    })
   424→
   425→    sock.on('connect', () => {
   426→      /* ready */
   427→    })
   428→
   429→    sock.on('connect_error', (err: { message: string }) => {
   430→      // Only show the error toast if we're actively trying to run code.
   431→      // During reconnection attempts, socket.io fires many of these
   432→      // and we don't want to spam the user.
   433→      if (!isRunningRef.current) return
   434→      toast.error('Cannot connect to runner', {
   435→        description: err.message || 'WebSocket connection failed',
   436→      })
   437→      setIsRunning(false)
   438→      isRunningRef.current = false
   439→    })
   440→
   441→    sock.on('started', () => {
   442→      // Server has spawned the python process
   443→      setAwaitingInput(false)
   444→    })
   445→
   446→    sock.on('output', (msg: { stream: string; data: string; promptLike?: boolean }) => {
   447→      const stream = msg.stream === 'stderr' ? 'stderr' : 'stdout'
   448→      const id = ++chunkIdRef.current
   449→      setChunks((prev) => [
   450→        ...prev,
   451→        { id, stream, text: msg.data, isPrompt: msg.promptLike },
   452→      ])
   453→      // If the server hints this looks like an input prompt (no trailing newline),
   454→      // focus the input bar.
   455→      if (msg.promptLike) {
   456→        setAwaitingInput(true)
   457→        setTimeout(() => inputRef.current?.focus(), 30)
   458→      } else if (stream === 'stdout') {
   459→        // A newline ended, so the prompt is resolved.
   460→        setAwaitingInput(false)
   461→      }
   462→    })
   463→
   464→    sock.on('image', (msg: { data: string; mime?: string }) => {
   465→      // Inline image (matplotlib figure rendered as PNG). The runner already
   466→      // base64-encoded the bytes — we wrap it as a data URL for <img src>.
   467→      const mime = msg.mime ?? 'image/png'
   468→      const src = `data:$\{mime};base64,$\{msg.data}`
   469→      const id = ++chunkIdRef.current
   470→      setChunks((prev) => [
   471→        ...prev,
   472→        { id, stream: 'image', text: '', src },
   473→      ])
   474→    })
   475→
   476→    sock.on('server', (msg: { port: number; host?: string }) => {
   477→      if (languageRef.current === 'flutter') {
   478→        // Set the port for the preview status panel
   479→        setFlutterPort(msg.port)
   480→        // Automatically open the Flutter app in a new browser tab.
   481→        // Use setTimeout to ensure the current page state is saved first
   482→        // and the window.open call happens in a clean event loop.
   483→        const port = msg.port
   484→        setTimeout(() => {
   485→          const previewUrl = window.location.origin + `/?XTransformPort=${port}`
   486→          window.open(previewUrl, '_blank', 'noopener,noreferrer')
   487→        }, 100)
   488→        toast.success('Flutter app is live!', {
   489→          description: `Opening preview in new tab (port ${port})...`,
   490→          duration: 4000,
   491→        })
   492→      } else {
   493→        const id = ++chunkIdRef.current
   494→        setChunks((prev) => [
   495→          ...prev,
   496→          {
   497→            id,
   498→            stream: 'server',
   499→            text: `Server started on port $\{msg.port}`,
   500→            port: msg.port,
   501→          },
   502→        ])
   503→        toast.success('Server started', {
   504→          description: `Listening on port $\{msg.port} — click the link in the console to open it.`,
   505→          duration: 8000,
   506→        })
   507→      }
   508→    })
   509→
   510→    sock.on('exit', (res: RunResult) => {
   511→      setResult(res)
   512→      setIsRunning(false)
   513→      isRunningRef.current = false
   514→      setAwaitingInput(false)
   515→      const ok =
   516→        res.code === 0 && !res.timedOut && !res.error
   517→      if (ok) {
   518→        toast.success('Program finished')
   519→      } else if (res.timedOut) {
   520→        toast.error('Timed out')
   521→      } else if (res.code !== null && res.code !== 0) {
   522→        toast.error(`Exited with code $\{res.code}`)
   523→      } else if (res.error) {
   524→        toast.error('Failed to run', { description: res.error })
   525→      }
   526→    })
   527→
   528→    sock.on('timeout', () => {
   529→      toast.error('Timed out')
   530→    })
   531→
   532→    socketRef.current = sock
   533→    return sock
   534→  }, [])
   535→
   536→  // ---- Auto-scroll console ----
   537→  useEffect(() => {
   538→    if (consoleEndRef.current) {
   539→      consoleEndRef.current.scrollIntoView({ block: 'end' })
   540→    }
   541→  }, [chunks])
   542→
   543→  // ---- Focus input when awaitingInput turns true ----
   544→  useEffect(() => {
   545→    if (awaitingInput && inputRef.current) {
   546→      inputRef.current.focus()
   547→    }
   548→  }, [awaitingInput])
   549→
   550→  // ---- Cleanup on unmount ----
   551→  useEffect(() => {
   552→    return () => {
   553→      socketRef.current?.disconnect()
   554→      socketRef.current = null
   555→    }
   556→  }, [])
   557→
   558→  // ---- Run handler ----
   559→  const handleRun = useCallback(() => {
   560→    if (isRunningRef.current) return
   561→    if (!code.trim()) {
   562→      toast.info('Nothing to run', {
   563→        description:
   564→          language === 'java' ? 'Write some Java first.' :
   565→          language === 'c' ? 'Write some C first.' :
   566→          language === 'cpp' ? 'Write some C++ first.' :
   567→          language === 'r' ? 'Write some R first.' :
   568→          language === 'javascript' ? 'Write some JavaScript first.' :
   569→          language === 'php' ? 'Write some PHP first.' :
   570→          language === 'csharp' ? 'Write some C# first.' :
   571→          language === 'dart' ? 'Write some Dart first.' :
   572→          language === 'flutter' ? 'Write some Flutter code first.' :
   573→          'Write some Python first.',
   574→      })
   575→      return
   576→    }
   577→    isRunningRef.current = true
   578→    setIsRunning(true)
   579→    setResult(null)
   580→    setChunks([])
   581→    setAwaitingInput(false)
   582→    setFlutterPort(null)
   583→    chunkIdRef.current = 0
   584→
   585→    const sock = ensureSocket()
   586→    const emitRun = () => sock.emit('run', { code, language, timeout: language === 'flutter' ? 120000 : 30000, stdin: stdinText })
   587→    if (sock.connected) {
   588→      emitRun()
   589→    } else {
   590→      // Wait for connection. The runner might be restarting (watchdog
   591→      // auto-restart), so give it extra time to come back up.
   592→      sock.once('connect', emitRun)
   593→      if (!sock.active) sock.connect()
   594→      // Safety timeout: if we never connect after 10s, surface the error
   595→      setTimeout(() => {
   596→        if (isRunningRef.current && !sock.connected) {
   597→          toast.error('Cannot connect to Python runner', {
   598→            description: 'The runner service may be restarting. Try again in a moment.',
   599→          })
   600→          isRunningRef.current = false
   601→          setIsRunning(false)
   602→        }

  1629→    input: 'bg-amber-500/20 text-amber-600 dark:text-amber-400',
  1630→  }[status.tone]
  1631→
  1632→  const dot = {
  1633→    idle: 'bg-muted-foreground/60',
  1634→    running: 'bg-amber-500 animate-pulse',
  1635→    success: 'bg-emerald-500',
  1636→    error: 'bg-rose-500',
  1637→    input: 'bg-amber-500 animate-pulse',
  1638→  }[status.tone]
  1639→
  1640→  const icon =
  1641→    status.tone === 'success' ? (
  1642→      <CircleCheck className="h-3 w-3" />
  1643→    ) : status.tone === 'error' ? (
  1644→      <CircleAlert className="h-3 w-3" />
  1645→    ) : status.tone === 'running' || status.tone === 'input' ? (
  1646→      <Loader2 className="h-3 w-3 animate-spin" />

  1648→
  1649→  return (
  1650→    <span

  1659→function ConsoleLine({ chunk }: { chunk: OutputChunk }) {
  1660→  // Image chunk (matplotlib figure rendered as inline PNG)
  1661→  if (chunk.stream === 'image' && chunk.src) {
  1662→    return (
  1663→      <span className="my-2 block">
  1664→        <img
  1665→          src={chunk.src}
  1666→          alt="matplotlib figure"
  1667→          className="max-w-full h-auto rounded-md border border-zinc-700/50 bg-white"
  1668→          style={{ display: 'block', maxHeight: '70vh' }}
  1669→        />
  1670→      </span>
  1671→    )
  1672→  }
  1673→
  1674→  // Server chunk: clickable link that opens the running server via Caddy.
  1675→  // The URL uses ?XTransformPort=<port> so the gateway routes the request
  1676→  // to the user's running Flask/Django/http.server app.
  1677→  if (chunk.stream === 'server' && chunk.port) {
  1678→    const port = chunk.port
  1679→    const href = `/?XTransformPort=$\{port}`
  1680→    return (
  1681→      <span
  1682→        className="my-2 block rounded-md border border-emerald-500/30 bg-emerald-500/10 px-3 py-2"
  1683→        style={{
  1684→          fontFamily:
  1685→            'var(--font-geist-mono), ui-monospace, SFMono-Regular, Menlo, monospace',
  1686→          whiteSpace: 'pre-wrap',
  1687→          overflowWrap: 'break-word',
  1688→        }}
  1689→      >
  1690→        <span className="text-emerald-400 font-semibold">▶ Server ready</span>
  1691→        <span className="text-zinc-300"> — listening on port </span>
  1692→        <span className="text-emerald-400 font-mono font-semibold">{port}</span>
  1693→        <span className="block mt-1">
  1694→          <a
  1695→            href={href}
  1696→            target="_blank"
  1697→            rel="noopener noreferrer"
  1698→            className="inline-flex items-center gap-1.5 text-emerald-400 hover:text-emerald-300 hover:underline underline-offset-2"
  1699→          >
  1700→            Open
  1701→            <span className="text-zinc-500">/?XTransformPort={port}</span>
  1702→            <svg
  1703→              className="h-3 w-3"
  1704→              viewBox="0 0 24 24"
  1705→              fill="none"
  1706→              stroke="currentColor"
  1707→              strokeWidth="2"
  1708→              strokeLinecap="round"
  1709→              strokeLinejoin="round"
  1710→            >
  1711→              <path d="M7 17L17 7M17 7H7M17 7v10" />
  1712→            </svg>
  1713→          </a>
  1714→        </span>
  1715→        <span className="block mt-1 text-xs text-zinc-500">
  1716→          Use Stop to terminate the server.
  1717→        </span>
  1718→      </span>
  1719→    )
  1720→  }
  1721→
  1722→  // Render with EXACT whitespace preservation.
  1723→  // white-space: pre-wrap preserves leading spaces, multiple consecutive
  1724→  // spaces, tabs, and line breaks exactly as the program produced them.
  1725→  // It also wraps long lines at the right edge instead of horizontal scroll.
  1726→  // overflow-wrap: break-word breaks very long unbreakable tokens (e.g.
  1727→  // long file paths in tracebacks) to prevent overflow.
  1728→  // tab-size: 4 renders tabs as 4 columns wide.
  1729→  const text = chunk.text
  1730→  const baseStyle: React.CSSProperties = {
  1731→    fontFamily:
  1732→      'var(--font-geist-mono), ui-monospace, SFMono-Regular, Menlo, monospace',
  1733→    whiteSpace: 'pre-wrap',
  1734→    overflowWrap: 'break-word',
  1735→    tabSize: 4,
  1736→  }
  1737→
  1738→  if (chunk.stream === 'input') {
  1739→    return (
  1740→      <span className="text-sky-400" style={baseStyle}>
  1741→        {text}
  1742→      </span>
  1743→    )
  1744→  }
  1745→  if (chunk.stream === 'stderr') {
  1746→    return (
  1747→      <span className="text-rose-400" style={baseStyle}>
  1748→        {text}
  1749→      </span>
  1750→    )
  1751→  }
  1752→  if (chunk.stream === 'system') {
  1753→    return (
  1754→      <span className="text-muted-foreground italic" style={baseStyle}>
  1755→        {text}
  1756→      </span>
  1757→    )
  1758→  }
  1759→  // stdout
  1760→  return (
  1761→    <span className="text-zinc-100" style={baseStyle}>
  1762→      {text}
  1763→    </span>
  1764→  )
  1765→}
